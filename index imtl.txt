<script>
const $ = id => document.getElementById(id);
const num = id => {
  const v = parseFloat($(id).value);
  return isNaN(v) ? 0 : v;
};

let lastResult = null;

function bindSeg(segId) {
  const seg = $(segId);
  seg.addEventListener('click', e => {
    const btn = e.target.closest('.seg-btn');
    if (!btn) return;
    seg.querySelectorAll('.seg-btn').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
    compute();
  });
}
bindSeg('sexSeg');
bindSeg('rateSeg');

['age','height','weight','bodyfat','activity','steps','targetWeight'].forEach(id => {
  $(id).addEventListener('input', compute);
  $(id).addEventListener('change', compute);
});
$('includeSteps').addEventListener('change', compute);

function compute() {
  const sex = document.querySelector('#sexSeg .seg-btn.active').dataset.val;
  const age = num('age');
  const height = num('height');
  const weight = num('weight');
  const bf = num('bodyfat');
  const factor = parseFloat($('activity').value);
  const steps = num('steps');
  const includeSteps = $('includeSteps').checked;
  const rate = parseFloat(document.querySelector('#rateSeg .seg-btn.active').dataset.rate);
  const targetWeight = num('targetWeight');

  const out = $('result');

  if (!age || !height || !weight) {
    out.innerHTML = '<div class="empty">✨ 填一下信息，看看数字长啥样</div>';
    lastResult = null;
    return;
  }

  let bmr, formula;
  if (bf >= 3 && bf <= 60) {
    const lbm = weight * (1 - bf / 100);
    bmr = 370 + 21.6 * lbm;
    formula = 'Katch-McArdle 公式';
  } else {
    bmr = sex === 'male'
      ? 10 * weight + 6.25 * height - 5 * age + 5
      : 10 * weight + 6.25 * height - 5 * age - 161;
    formula = 'Mifflin-St Jeor 公式';
  }

  const baseTdee = bmr * factor;
  const stepKcal = steps * weight * 0.0005;
  const tdee = includeSteps ? baseTdee + stepKcal : baseTdee;

  const targetDeficit = rate * 7700 / 7;
  const floor = sex === 'male' ? 1500 : 1200;
  let intake = tdee - targetDeficit;
  let adjusted = false;
  if (intake < floor) { intake = floor; adjusted = true; }

  const actualDeficit = tdee - intake;
  const weekly = actualDeficit * 7 / 7700;
  const monthly = actualDeficit * 30 / 7700;

  let warns = '';
  if (adjusted) {
    warns += `<div class="warn">⚠️ 按这个速度，摄入会低到不太现实，已经自动拉回安全线 <b>${floor} kcal</b>。数字好看，身体不一定买账。</div>`;
  } else if (intake < bmr) {
    warns += `<div class="warn">⚠️ 建议摄入（${Math.round(intake)} kcal）低于基础代谢（${Math.round(bmr)} kcal），长期这么吃，代谢和肌肉可能会抗议。</div>`;
  }
  if (rate >= 0.75) {
    warns += `<div class="warn">📉 选了较快的速度。减得快，掉肌肉和反弹的概率也高，建议悠着点。</div>`;
  }

  let stepNote = '';
  if (steps > 0) {
    stepNote = `<div class="info">🚶 每天 ${steps.toLocaleString()} 步，大约多烧 <b>${Math.round(stepKcal)} kcal</b>。${includeSteps ? '已计入总消耗。' : '未计入总消耗。'}</div>`;
  }

  let goalNote = '';
  if (targetWeight > 0 && targetWeight < weight && weekly > 0) {
    const needKg = weight - targetWeight;
    const days = needKg / weekly * 7;
    const weeks = days / 7;
    goalNote = `<div class="info">🎯 从 ${weight}kg 到 ${targetWeight}kg，差 ${needKg.toFixed(1)}kg。<br>按当前速度大约 <b>${weeks.toFixed(1)} 周</b>（${Math.round(days)} 天）。当然，现实不会这么听话。</div>`;
  } else if (targetWeight > 0 && targetWeight >= weight) {
    goalNote = `<div class="info">🎯 目标体重不小于当前体重，这工具只会算减脂哦。</div>`;
  }

  out.innerHTML = `
    <div class="stat-grid">
      <div class="stat">
        <div class="lbl">基础代谢 BMR</div>
        <div class="val">${Math.round(bmr)}<span>kcal</span></div>
        <div class="tag">${formula}</div>
      </div>
      <div class="stat">
        <div class="lbl">每日总消耗 TDEE</div>
        <div class="val">${Math.round(tdee)}<span>kcal</span></div>
        <div class="tag">活动系数 ${factor}</div>
      </div>
    </div>

    <div class="highlight">
      <div class="hl-label">建议每日摄入 · 仅供参考</div>
      <div class="hl-value">${Math.round(intake)}<span>kcal</span></div>
      <div class="hl-sub">每日热量缺口约 ${Math.round(actualDeficit)} kcal</div>
    </div>

    <div class="stat-grid">
      <div class="stat">
        <div class="lbl">预计每周减</div>
        <div class="val">${weekly.toFixed(2)}<span>kg</span></div>
      </div>
      <div class="stat">
        <div class="lbl">预计每月减</div>
        <div class="val">${monthly.toFixed(2)}<span>kg</span></div>
      </div>
    </div>

    ${stepNote}
    ${goalNote}
    ${warns}
    <div class="info">🎲 以上都是公式算出来的理想值，真实身体还会看水分、激素、睡眠、压力和心情。</div>
  `;

  lastResult = {
    bmr: Math.round(bmr),
    tdee: Math.round(tdee),
    intake: Math.round(intake),
    deficit: Math.round(actualDeficit),
    weekly: weekly.toFixed(2),
    monthly: monthly.toFixed(2)
  };
}

$('copyBtn').addEventListener('click', () => {
  if (!lastResult) { alert('先填信息算一下呀'); return; }
  const r = lastResult;
  const text =
`【热量小算盘 · 娱乐版】
基础代谢 BMR：${r.bmr} kcal
每日总消耗：${r.tdee} kcal
建议每日摄入：${r.intake} kcal
每日热量缺口：${r.deficit} kcal
预计每周减：${r.weekly} kg
预计每月减：${r.monthly} kg
（公式估算，纯娱乐，仅供参考）`;

  navigator.clipboard.writeText(text).then(
    () => alert('已复制，去粘贴吧～'),
    () => alert('复制失败，手动长按选择吧')
  );
});

$('resetBtn').addEventListener('click', () => {
  $('age').value = 30;
  $('height').value = 160;
  $('weight').value = 65;
  $('bodyfat').value = '';
  $('steps').value = 8000;
  $('targetWeight').value = '';
  $('activity').value = '1.375';
  $('includeSteps').checked = false;
  document.querySelectorAll('#sexSeg .seg-btn').forEach((b, i) => b.classList.toggle('active', i === 0));
  document.querySelectorAll('#rateSeg .seg-btn').forEach((b, i) => b.classList.toggle('active', i === 1));
  compute();
});

compute();
</script>
